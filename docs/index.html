Axiom Works AgentReader

Quick start

# Stdio MCP (Claude Desktop / Cursor / Claude Code)
npx -y @axiomworks/agentreader-mcp
# set AGENTREADER_PRIVATE_KEY=0x… for auto x402 USDC payment on Base

# Typed SDK
npm i @axiomworks/agentreader
import { createClient } from "@axiomworks/agentreader";
const client = createClient({ privateKey: process.env.AGENTREADER_PRIVATE_KEY });
await client.readUrl({ url: "https://example.com" });

# Python (LangChain / CrewAI / LlamaIndex / PydanticAI / OpenAI Agents)
pip install agentreader-tools
# pip install agentreader-crewai llama-index-tools-agentreader "agentreader-agents[all]"
from agentreader_tools import agentreader_tools
tools = agentreader_tools(private_key=KEY, max_price_usd=0.01)

# Or low-level @x402/fetch
npm i @x402/fetch @x402/evm viem
await pay("https://api.axiomworks.co.uk/read?url=https://example.com")
